cmake_minimum_required(VERSION 3.20)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/version.json")
    set(GINCY_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/version.json")
else()
    set(GINCY_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../version.json")
endif()
file(READ "${GINCY_VERSION_FILE}" GINCY_VERSION_JSON)
string(JSON GINCY_PRODUCT GET "${GINCY_VERSION_JSON}" product)
string(JSON GINCY_FRAMEWORK_API GET "${GINCY_VERSION_JSON}" framework_api)
string(JSON GINCY_NATIVE_ABI GET "${GINCY_VERSION_JSON}" native_abi)
string(JSON GINCY_PACKAGE_FORMAT GET "${GINCY_VERSION_JSON}" package_format)
string(JSON GINCY_NETWORK_PROTOCOL GET "${GINCY_VERSION_JSON}" network_protocol)
string(JSON GINCY_MANAGEMENT_API GET "${GINCY_VERSION_JSON}" management_api)
string(JSON GINCY_CONTENT_SCHEMA_API GET "${GINCY_VERSION_JSON}" content_schema_api)
string(JSON GINCY_CHANNEL GET "${GINCY_VERSION_JSON}" channel)
project(gincy_core VERSION ${GINCY_PRODUCT} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
option(GINCY_STATIC_RUNTIME "Static MSVC CRT and static OpenSSL (use with vcpkg *-windows-static)" OFF)
option(GINCY_DELAYLOAD_LIBPQ "Delay-load libpq.dll so the GMod module starts without PostgreSQL DLLs" ON)
set(GINCY_EXPECT_ARCH "" CACHE STRING "If set to 32 or 64, fail configure when pointer width does not match")
if(GINCY_STATIC_RUNTIME AND MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(GINCY_STATIC_RUNTIME)
    set(OPENSSL_USE_STATIC_LIBS ON)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(GINCY_ARCH 64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set(GINCY_ARCH 32)
else()
    message(FATAL_ERROR "Unsupported pointer width")
endif()
if(GINCY_EXPECT_ARCH STREQUAL "32" OR GINCY_EXPECT_ARCH STREQUAL "64")
    if(NOT GINCY_ARCH EQUAL ${GINCY_EXPECT_ARCH})
        message(FATAL_ERROR "Refusing to build: this generator is ${GINCY_ARCH}-bit but GINCY_EXPECT_ARCH=${GINCY_EXPECT_ARCH}")
    endif()
endif()
find_package(PostgreSQL REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL 1.1.1 REQUIRED)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/version.hpp.in" "${CMAKE_CURRENT_BINARY_DIR}/gincy_version.hpp" @ONLY)
add_library(gincy_package STATIC src/package.cpp src/json.cpp src/gvm.cpp)
target_include_directories(gincy_package PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(gincy_package PUBLIC OpenSSL::Crypto)
if(WIN32)
    target_compile_definitions(gincy_package PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/gmod-module-base/include/GarrysMod/Lua/Interface.h")
    set(FETCHCONTENT_SOURCE_DIR_GMOD_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/gmod-module-base")
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/gmod-module-base/include/GarrysMod/Lua/Interface.h")
    set(FETCHCONTENT_SOURCE_DIR_GMOD_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/gmod-module-base")
endif()
include(FetchContent)
FetchContent_Declare(gmod_headers
    GIT_REPOSITORY https://github.com/Facepunch/gmod-module-base.git
    GIT_TAG e0156e5f96cd0bad8eed738f0d20e2c9a14b2a0d
    SOURCE_SUBDIR include
)
FetchContent_GetProperties(gmod_headers)
if(NOT gmod_headers_POPULATED)
    FetchContent_MakeAvailable(gmod_headers)
endif()
add_library(gincy_pool STATIC src/pool.cpp)
target_link_libraries(gincy_pool PUBLIC PostgreSQL::PostgreSQL Threads::Threads)
target_include_directories(gincy_pool PUBLIC src)
add_library(gincy_http STATIC src/http.cpp)
target_include_directories(gincy_http PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(gincy_http PUBLIC gincy_package)
if(WIN32)
    target_link_libraries(gincy_pool PUBLIC ws2_32)
    target_link_libraries(gincy_http PUBLIC ws2_32)
    target_compile_definitions(gincy_pool PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_definitions(gincy_http PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
add_library(gincy_core SHARED src/module.cpp)
target_include_directories(gincy_core PRIVATE ${gmod_headers_SOURCE_DIR}/include "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(gincy_core PRIVATE gincy_pool gincy_package gincy_http)
if(WIN32)
    target_compile_definitions(gincy_core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(GINCY_DELAYLOAD_LIBPQ)
        target_link_libraries(gincy_core PRIVATE delayimp)
        target_link_options(gincy_core PRIVATE "/DELAYLOAD:libpq.dll")
    endif()
else()
    target_compile_options(gincy_pool PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(gincy_core PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
    target_compile_options(gincy_http PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    set(GINCY_PLATFORM win)
else()
    set(GINCY_PLATFORM linux)
endif()
if(GINCY_ARCH EQUAL 64)
    set(GINCY_SUFFIX "_${GINCY_PLATFORM}64.dll")
elseif(WIN32)
    set(GINCY_SUFFIX "_win32.dll")
else()
    set(GINCY_SUFFIX "_linux.dll")
endif()
set_target_properties(gincy_core PROPERTIES PREFIX "gmsv_" SUFFIX "${GINCY_SUFFIX}")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../garrysmod/lua/bin")
    set(GINCY_DEPLOY_BIN "${CMAKE_CURRENT_SOURCE_DIR}/../garrysmod/lua/bin")
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../garrysmod/lua/bin")
    set(GINCY_DEPLOY_BIN "${CMAKE_CURRENT_SOURCE_DIR}/../../garrysmod/lua/bin")
endif()
if(GINCY_DEPLOY_BIN)
    add_custom_command(TARGET gincy_core POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${GINCY_DEPLOY_BIN}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:gincy_core> "${GINCY_DEPLOY_BIN}/$<TARGET_FILE_NAME:gincy_core>"
        COMMENT "Deploy gincy_core to garrysmod/lua/bin")
endif()
add_executable(gincy src/cli.cpp)
target_link_libraries(gincy PRIVATE gincy_package)
if(WIN32)
    target_compile_definitions(gincy PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(NOT WIN32)
    target_link_libraries(gincy PRIVATE pthread)
endif()
include(CTest)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/gvm_test.cpp")
    set(GINCY_TESTS "${CMAKE_CURRENT_SOURCE_DIR}/tests")
else()
    set(GINCY_TESTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests")
endif()
if(BUILD_TESTING)
    add_executable(package_decrypt "${GINCY_TESTS}/package_decrypt.cpp")
    target_link_libraries(package_decrypt PRIVATE gincy_package)
    add_executable(package_verify "${GINCY_TESTS}/package_verify.cpp")
    target_link_libraries(package_verify PRIVATE gincy_package)
    add_executable(pool_test "${GINCY_TESTS}/pool_test.cpp")
    target_link_libraries(pool_test PRIVATE gincy_pool gincy_package)
    add_executable(gvm_test "${GINCY_TESTS}/gvm_test.cpp")
    target_link_libraries(gvm_test PRIVATE gincy_package)
    add_executable(package_v2_test "${GINCY_TESTS}/package_v2_test.cpp")
    target_link_libraries(package_v2_test PRIVATE gincy_package)
    add_test(NAME gvm COMMAND gvm_test)
    add_test(NAME package_v2 COMMAND package_v2_test)
    add_test(NAME pool COMMAND pool_test)
endif()
install(TARGETS gincy_core LIBRARY DESTINATION garrysmod/lua/bin RUNTIME DESTINATION garrysmod/lua/bin)
install(TARGETS gincy RUNTIME DESTINATION tools)
